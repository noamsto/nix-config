# nix-amd-ai Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a self-contained Nix flake (`noamsto/nix-amd-ai`) that packages the AMD AI inference stack (XRT, xrt-plugin-amdxdna, FastFlowLM, Lemonade) from source with a NixOS module and CI caching.

**Architecture:** Flake-parts flake exposing 4 packages, an overlay, and a NixOS module. Packages adapt existing derivations from `robcohen/nix-amd-npu` where available. CI builds and pushes to Cachix.

**Tech Stack:** Nix flakes, flake-parts, CMake/Ninja (C++), Cargo (Rust), GitHub Actions, Cachix

**Spec:** `docs/superpowers/specs/2026-04-16-nix-amd-ai-design.md` (in nix-config repo)

**Reference derivations:** `github:robcohen/nix-amd-npu` pkgs/ directory — adapt, don't copy blindly.

### Task 6: Create NixOS module

Port and improve `nixos/hardware/amd-npu.nix` + `nixos/services/lemonade.nix` from nix-config.

**Files:**
- Create: `modules/amd-npu.nix`

- [ ] **Step 1: Write modules/amd-npu.nix**

Combine the hardware module and lemonade service into one file. Key improvements over the nix-config version:
- Self-contained — references packages from this flake, not `pkgs.xrt`
- `xrt-combined` merged tree built internally
- ROCm option for lemonade service
- `LimitMEMLOCK=infinity` on lemond service
- Automatic Cachix substituter configuration

```nix
{
  config,
  lib,
  pkgs,
  ...
}:
with lib; let
  cfg = config.hardware.amd-npu;
  xrtPrefix = "${pkgs.xrt}/opt/xilinx/xrt";

  xrt-combined = pkgs.runCommand "xrt-combined" {} ''
    mkdir -p $out
    cp -rs ${xrtPrefix}/* $out/
    chmod -R u+w $out/lib
    ln -sf ${pkgs.xrt-plugin-amdxdna}/lib/libxrt_driver_xdna* $out/lib/
  '';
in {
  options.hardware.amd-npu = {
    enable = mkEnableOption "AMD XDNA NPU support for AI inference";

    enableFastFlowLM = mkOption {
      type = types.bool;
      default = true;
      description = "Install FastFlowLM for LLM inference on NPU.";
    };

    enableLemonade = mkOption {
      type = types.bool;
      default = true;
      description = "Enable Lemonade server (OpenAI-compatible local AI API).";
    };

    enableROCm = mkOption {
      type = types.bool;
      default = false;
      description = "Enable ROCm GPU support for Lemonade backends (llamacpp:rocm).";
    };

    lemonade = {
      port = mkOption {
        type = types.port;
        default = 13305;
        description = "HTTP port for the Lemonade API.";
      };
      host = mkOption {
        type = types.str;
        default = "localhost";
        description = "Address to bind for connections.";
      };
      user = mkOption {
        type = types.str;
        description = "User to run lemond as (must be in video/render groups).";
      };
    };
  };

  config = mkIf cfg.enable {
    assertions = [
      {
        assertion = versionAtLeast config.boot.kernelPackages.kernel.version "6.14";
        message = "AMD NPU requires kernel 6.14+. Current: ${config.boot.kernelPackages.kernel.version}";
      }
    ];

    boot.kernelParams = ["iommu.passthrough=0"];
    boot.kernelModules = ["amdxdna"];

    services.udev.extraRules = ''
      SUBSYSTEM=="accel", KERNEL=="amdxdna", MODE="0660", GROUP="video"
      SUBSYSTEM=="misc", KERNEL=="amdxdna", MODE="0660", GROUP="video"
    '';

    security.pam.loginLimits = [
      { domain = "@video"; type = "soft"; item = "memlock"; value = "unlimited"; }
      { domain = "@video"; type = "hard"; item = "memlock"; value = "unlimited"; }
      { domain = "@render"; type = "soft"; item = "memlock"; value = "unlimited"; }
      { domain = "@render"; type = "hard"; item = "memlock"; value = "unlimited"; }
    ];

    environment.variables = {
      "XILINX_XRT" = "${xrt-combined}";
      "XRT_PATH" = "${xrt-combined}";
    };

    environment.systemPackages =
      [pkgs.pciutils pkgs.lshw xrt-combined]
      ++ optionals cfg.enableFastFlowLM [pkgs.fastflowlm]
      ++ optionals cfg.enableROCm [pkgs.rocmPackages.clr];

    # Lemonade service
    systemd.services.lemond = mkIf cfg.enableLemonade {
      description = "Lemonade AI Server";
      after = ["network-online.target"];
      wants = ["network-online.target"];
      wantedBy = ["multi-user.target"];

      serviceConfig = {
        Type = "simple";
        User = cfg.lemonade.user;
        ExecStart = "${pkgs.lemonade}/bin/lemond --port ${toString cfg.lemonade.port} --host ${cfg.lemonade.host}";
        Restart = "on-failure";
        RestartSec = "5s";
        KillSignal = "SIGINT";
        LimitMEMLOCK = "infinity";
        Environment = let
          rocmLibs = optionalString cfg.enableROCm ":${pkgs.rocmPackages.clr}/lib";
        in [
          "XILINX_XRT=${xrt-combined}"
          "XRT_PATH=${xrt-combined}"
          "LD_LIBRARY_PATH=${xrt-combined}/lib${rocmLibs}"
          "PATH=${makeBinPath ([xrt-combined] ++ optionals cfg.enableFastFlowLM [pkgs.fastflowlm])}:/run/current-system/sw/bin"
        ];
      };
    };
  };
}
```

**Note:** The module references `pkgs.xrt`, `pkgs.xrt-plugin-amdxdna`, etc. Consumers must apply the overlay from this flake for these to resolve. The `nixosModules.default` import should handle this — update `flake.nix` to pass the overlay automatically:

Update the `flake` section in `flake.nix` accordingly.

```yaml
name: Check for upstream updates

on:
  schedule:
    - cron: '0 6 * * 1'  # Weekly Monday 6am UTC
  workflow_dispatch:

jobs:
  check-updates:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: DeterminateSystems/nix-installer-action@main

      - name: Check for new releases
        id: check
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          FLM_LATEST=$(gh api repos/FastFlowLM/FastFlowLM/releases/latest --jq '.tag_name' | sed 's/^v//')
          LEM_LATEST=$(gh api repos/lemonade-sdk/lemonade/releases/latest --jq '.tag_name' | sed 's/^v//')
          XDNA_LATEST=$(gh api "repos/amd/xdna-driver/commits?sha=1.7&per_page=1" --jq '.[0].sha')

          FLM_CURRENT=$(grep 'version = ' pkgs/fastflowlm/default.nix | head -1 | sed 's/.*"\(.*\)".*/\1/')
          LEM_CURRENT=$(grep 'version = ' pkgs/lemonade/default.nix | head -1 | sed 's/.*"\(.*\)".*/\1/')
          XDNA_CURRENT=$(grep 'rev = ' pkgs/xrt-plugin-amdxdna/default.nix | head -1 | sed 's/.*"\(.*\)".*/\1/')

          echo "flm_update=$( [ "$FLM_LATEST" != "$FLM_CURRENT" ] && echo true || echo false )" >> "$GITHUB_OUTPUT"
          echo "lem_update=$( [ "$LEM_LATEST" != "$LEM_CURRENT" ] && echo true || echo false )" >> "$GITHUB_OUTPUT"
          echo "xdna_update=$( [ "$XDNA_LATEST" != "$XDNA_CURRENT" ] && echo true || echo false )" >> "$GITHUB_OUTPUT"

          echo "FLM: $FLM_CURRENT -> $FLM_LATEST"
          echo "Lemonade: $LEM_CURRENT -> $LEM_LATEST"
          echo "XDNA: $XDNA_CURRENT -> $XDNA_LATEST"

      - name: Create update issue
        if: steps.check.outputs.flm_update == 'true' || steps.check.outputs.lem_update == 'true' || steps.check.outputs.xdna_update == 'true'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh issue create \
            --title "Upstream updates available" \
            --body "New versions detected by automated check. Update package derivations and hashes."
```
